    <div id="setup">

        <h1>Create a Proposal</h1>

        <label><input type="checkbox" id="unclickableNo"> Make 'No' unclickable?</label>

        <button class="btn" id="yesBtn" onclick="generateLink()">Generate Link</button>

        <h1 id="displayMessage">Will you be my Valentine?</h1>

            <button class="btn" id="yesBtn" onclick="celebrate()">Yes</button>

            <button class="btn" id="noBtn">No</button>

<script>

    const urlParams = new URLSearchParams(window.location.search);

    const vName = urlParams.get('name');

    const restrictNo = urlParams.get('no') === 'true';


    // Logic to switch views

    if (vName) {

        document.getElementById('setup').classList.add('hidden');

        document.getElementById('proposal').classList.remove('hidden');

        document.getElementById('displayMessage').innerText = `${vName}, will you be my Valentine? ❤️`;

        

        const noBtn = document.getElementById('noBtn');

        if (restrictNo) {

            noBtn.onmouseover = () => {

                const x = Math.random() * (window.innerWidth - noBtn.offsetWidth);

                const y = Math.random() * (window.innerHeight - noBtn.offsetHeight);

                noBtn.style.position = 'absolute';

                noBtn.style.left = x + 'px';

                noBtn.style.top = y + 'px';

            };

        }

    }


    function generateLink() {

        const name = document.getElementById('valentineName').value;

        const isUnclickable = document.getElementById('unclickableNo').checked;

        if (!name) return alert("Please enter a name!");

        

        const baseUrl = window.location.origin + window.location.pathname;

        const finalUrl = `${baseUrl}?name=${encodeURIComponent(name)}&no=${isUnclickable}`;

        

        const linkDisplay = document.getElementById('resultLink');

        linkDisplay.innerHTML = `Share this link: <br><a href="${finalUrl}">${finalUrl}</a>`;

    }


    function celebrate() {

        document.querySelector('.container').innerHTML = "<h1>Yay! See you on the 14th! 🌹❤️</h1>";

    }

</script>
